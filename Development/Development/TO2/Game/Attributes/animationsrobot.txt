[transitions]

// transitions are formatted as follows:
//
// "animation name" movement_type { initial set }  +{ add set }  ~{ remove set }  ={ constant set }  !{ not set }
//
// movement_type is one of: None, Walk, Run, Swim, Climb, Encode_NG, Encode_G
//
// to go from a set of props F to a set of props T, via the transition, the following
// conditions must be met:
//
// F must contain all elements of "initial set"
// F must have none of "add set"
// T must have all of "add set"
// F must have all of "remove set"
// T must not have "remove set"
// both F and T must have all of "constant set"
// both F and T must not have anything in "not set"
//
// more specific transitions must be specified later (see 1StGd-1StGuAiIn vs. 1StGd-1StGuAi)
